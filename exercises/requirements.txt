# Everything the exercises need, pinned so every learner gets the same results.
# Install once from the repository root: pip install -r exercises/requirements.txt
#
# pytest runs every exercise's tests.
# Django, pytest-django and Django REST Framework are for the Django exercises
# (exercises/backend/dj_*), which run offline on an in-memory SQLite database.
# drf-spectacular is for the API documentation exercise (exercises/backend/dj_api_docs),
# which builds the OpenAPI schema in-process.
# Celery, redis-py and fakeredis are for the Celery and Redis exercises
# (exercises/backend/dj_celery, dj_redis): Celery runs the tasks eagerly in the test
# process and fakeredis stands in for the Redis server, so neither needs a broker or server.
# factory_boy and Faker are for the test-data exercise (exercises/backend/dj_factory_boy).
# The CI/CD exercise (exercises/backend/dj_cicd) reads its workflow with PyYAML,
# which drf-spectacular installs.
# Channels is for the WebSockets exercise (exercises/backend/dj_channels): its tests open
# sockets with WebsocketCommunicator on the in-memory channel layer, in the test process.
# Its testing module imports Daphne, the ASGI server, so the daphne extra comes too.
# Django 5.2 is the long-term-support release and runs on Python 3.10 to 3.14.
pytest==9.1.1
Django==5.2.19
pytest-django==4.14.0
djangorestframework==3.18.3
drf-spectacular==0.30.0
celery==5.6.3
redis==8.1.0
fakeredis==2.40.0
factory_boy==3.3.3
Faker==40.44.0
channels[daphne]==4.3.2
daphne==4.2.3
# pyspark and delta-spark are for the Databricks exercises (exercises/databricks/*): they run
# on a local Spark session with Delta Lake switched on, with no Databricks workspace.
# Spark needs Java 17 or newer installed; delta-spark 4.0.1 is the Delta release built for Spark 4.0.
# The first run downloads the Delta jars once from Maven Central, then it works offline.
pyspark==4.0.5
delta-spark==4.0.1
# pandas and pyarrow are pyspark's own sql extra, for the UDF exercise (exercises/databricks/advanced_topics):
# a pandas UDF gets its rows as pandas Series in Apache Arrow batches.
# pyspark 4.0 needs pandas 2.0+ and pyarrow 11+; both releases have wheels for Python 3.10 to 3.12.
pandas==2.3.3
pyarrow==21.0.0
# duckdb is for the Data Engineering exercises (exercises/data_engineering/*) that run SQL
# and for the SQL exercises (exercises/sql/*):
# each test opens an in-process DuckDB database in memory (and writes Parquet to a temporary folder
# for the data skipping one), so there is no server to start. The others need only the standard library.
# duckdb 1.5.6 has wheels for Python 3.10 to 3.14.
duckdb==1.5.6
# sqlglot is for the SQL exercises (exercises/sql/*): it reads the learner's T-SQL with its T-SQL dialect
# and writes it as DuckDB SQL, so the tests run T-SQL with no SQL Server. Pure Python, for Python 3.9+.
sqlglot==30.24.0
